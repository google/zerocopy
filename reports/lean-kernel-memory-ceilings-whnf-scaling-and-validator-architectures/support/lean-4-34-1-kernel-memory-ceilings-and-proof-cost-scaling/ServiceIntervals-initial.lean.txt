abbrev State :=
  (Nat × Bool × Nat) × Nat × Nat × Nat × Bool × Bool × List (Nat × Nat) × Option Bool × Nat × Nat

noncomputable def baseline (callers : List Nat) : List (Nat × Nat) :=
  (0, 0) :: callers.map (fun k => (1, k))

noncomputable def initialState (callers : List Nat) : State :=
  ((0, false, 0), 0, 0, 0, false, false, baseline callers, none, 0, 0)

noncomputable def occupiedCount (s : State) : Nat :=
  if s.2.2.2.2.1 then 1 else 0

def IsBusy (p : Nat × Bool × Nat) : Prop :=
  p = (1, false, 0) ∨ p = (2, false, 0) ∨ p = (3, false, 0) ∨
  (∃ o, p = (4, o, 0)) ∨ (∃ k, p = (5, false, k)) ∨ (∃ o, p = (6, o, 0)) ∨
  (∃ o, p = (7, o, 0)) ∨ (∃ o, p = (8, o, 0)) ∨ (∃ o, p = (9, o, 0)) ∨
  (∃ o, p = (10, o, 0))

def Bad (s : State) : Prop :=
  ∃ reason, s.1 = (11, false, reason)

def ServiceStep (lbl : Nat × Nat × Nat × Bool) (s s' : State) : Prop :=
  let ph := s.1; let d := s.2.1; let n := s.2.2.1; let rem := s.2.2.2.1
  let occ := s.2.2.2.2.1; let own := s.2.2.2.2.2.1; let stk := s.2.2.2.2.2.2.1
  let rep := s.2.2.2.2.2.2.2.1; let wk := s.2.2.2.2.2.2.2.2.1; let el := s.2.2.2.2.2.2.2.2.2
  (ph = (1, false, 0) ∧ lbl = (3, 0, 0, false) ∧
    s' = ((2, false, 0), d, n, rem, occ, own, (2, 0) :: stk, rep, wk + 1, el + 1)) ∨
  (ph = (2, false, 0) ∧ lbl = (4, 0, 0, false) ∧
    s' = ((3, false, 0), d, n, rem, occ, own, (3, 0) :: stk, rep, wk + 1, el + 1)) ∨
  ((∃ k, ph = (3, false, 0) ∧ rem = k + 1 ∧ occ = false ∧ lbl = (5, 0, 0, false) ∧
      s' = ((3, false, 0), d, n, k, true, true, stk, rep, wk + 1, el + 1)) ∨
    (ph = (3, false, 0) ∧ rem = 0 ∧ lbl = (6, 0, 0, true) ∧
      s' = ((4, true, 0), d, n, rem, occ, own, stk, rep, wk + 1, el + 1)) ∨
    (∃ k, ph = (3, false, 0) ∧ rem = k + 1 ∧ occ = true ∧ lbl = (6, 0, 0, false) ∧
      s' = ((4, false, 0), d, n, rem, occ, own, stk, rep, wk + 1, el + 1))) ∨
  ((∃ rest, ph = (4, true, 0) ∧ stk = (3, 0) :: rest ∧ n = 0 ∧ lbl = (7, 0, 0, true) ∧
      s' = ((6, true, 0), d, n, rem, occ, own, rest, rep, wk + 1, el + 1)) ∨
    (∃ k rest, ph = (4, true, 0) ∧ stk = (3, 0) :: rest ∧ n = k + 1 ∧ lbl = (7, 0, 0, true) ∧
      s' = ((5, false, k), d, n, rem, occ, own, rest, rep, wk + 1, el + 1)) ∨
    (∃ rest, ph = (4, false, 0) ∧ stk = (3, 0) :: rest ∧ lbl = (7, 0, 0, false) ∧
      s' = ((6, false, 0), d, n, rem, occ, own, rest, rep, wk + 1, el + 1)) ∨
    (∃ o, ph = (4, o, 0) ∧ stk.head? ≠ some (3, 0) ∧ lbl = (15, 0, 0, false) ∧
      s' = ((11, false, 2), d, n, rem, occ, own, stk, rep, wk + 1, el + 1))) ∨
  ((ph = (5, false, 0) ∧ lbl = (8, 0, 0, false) ∧
      s' = ((6, true, 0), d, n, rem, occ, own, stk, rep, wk + 1, el + 1)) ∨
    (∃ k, ph = (5, false, k + 1) ∧ lbl = (8, 0, 0, false) ∧
      s' = ((5, false, k), d, n, rem, occ, own, stk, rep, wk + 1, el + 1))) ∨
  ((∃ o, ph = (6, o, 0) ∧ own = true ∧ lbl = (9, 0, 0, o) ∧
      s' = ((7, o, 0), d, n, rem, occ, own, (4, 0) :: stk, rep, wk + 1, el + 1)) ∨
    (∃ o, ph = (6, o, 0) ∧ own = false ∧ lbl = (9, 0, 0, o) ∧
      s' = ((8, o, 0), d, n, rem, occ, own, (4, 0) :: stk, rep, wk + 1, el + 1))) ∨
  ((∃ o, ph = (7, o, 0) ∧ own = true ∧ occ = true ∧ lbl = (10, 0, 0, o) ∧
      s' = ((8, o, 0), d, n, rem, false, false, stk, rep, wk + 1, el + 1)) ∨
    (∃ o, ph = (7, o, 0) ∧ own = false ∧ lbl = (11, 0, 0, o) ∧
      s' = ((11, false, 0), d, n, rem, occ, own, stk, rep, wk + 1, el + 1)) ∨
    (∃ o, ph = (7, o, 0) ∧ own = true ∧ occ = false ∧ lbl = (11, 1, 0, o) ∧
      s' = ((11, false, 1), d, n, rem, occ, own, stk, rep, wk + 1, el + 1))) ∨
  ((∃ o rest, ph = (8, o, 0) ∧ stk = (4, 0) :: rest ∧ lbl = (12, 0, 0, o) ∧
      s' = ((9, o, 0), d, n, rem, occ, own, rest, rep, wk + 1, el + 1)) ∨
    (∃ o, ph = (8, o, 0) ∧ stk.head? ≠ some (4, 0) ∧ lbl = (15, 0, 0, false) ∧
      s' = ((11, false, 2), d, n, rem, occ, own, stk, rep, wk + 1, el + 1))) ∨
  (∃ o, ph = (9, o, 0) ∧ lbl = (13, 0, 0, o) ∧
    s' = ((10, o, 0), d, n, rem, occ, own, stk, some o, wk + 1, el + 1)) ∨
  ((∃ o rest, ph = (10, o, 0) ∧ stk = (2, 0) :: rest ∧ lbl = (14, 0, 0, o) ∧
      s' = ((0, false, 0), d, n, rem, occ, own, rest, rep, wk + 1, el + 1)) ∨
    (∃ o, ph = (10, o, 0) ∧ stk.head? ≠ some (2, 0) ∧ lbl = (15, 0, 0, false) ∧
      s' = ((11, false, 2), d, n, rem, occ, own, stk, rep, wk + 1, el + 1)))

def LabeledStep (lbl : Nat × Nat × Nat × Bool) (s s' : State) : Prop :=
  (∃ d n, lbl = (0, d, n, false) ∧ s.1 = (0, false, 0) ∧
    s' = ((1, false, 0), d, n, d, s.2.2.2.2.1, s.2.2.2.2.2.1, s.2.2.2.2.2.2.1, none,
      s.2.2.2.2.2.2.2.2.1, s.2.2.2.2.2.2.2.2.2 + 1)) ∨
  (lbl = (1, 0, 0, false) ∧ s.1 = (0, false, 0) ∧
    s' = (s.1, s.2.1, s.2.2.1, s.2.2.2.1, s.2.2.2.2.1, s.2.2.2.2.2.1, s.2.2.2.2.2.2.1,
      s.2.2.2.2.2.2.2.1, s.2.2.2.2.2.2.2.2.1, s.2.2.2.2.2.2.2.2.2 + 1)) ∨
  (lbl = (2, 0, 0, false) ∧ IsBusy s.1 ∧
    s' = (s.1, s.2.1, s.2.2.1, s.2.2.2.1, s.2.2.2.2.1, s.2.2.2.2.2.1, s.2.2.2.2.2.2.1,
      s.2.2.2.2.2.2.2.1, s.2.2.2.2.2.2.2.2.1, s.2.2.2.2.2.2.2.2.2 + 1)) ∨
  ServiceStep lbl s s'

def Step (s s' : State) : Prop :=
  ∃ lbl, LabeledStep lbl s s'

/-- Audit 1: Work/elapsed monotonicity and declared per-edge increments. -/
theorem global_counter_accounting :
    (∀ {lbl s s'}, ServiceStep lbl s s' →
      3 ≤ lbl.1 ∧ ¬(lbl.1 = 1 ∨ lbl.1 = 2) ∧ s.1 ≠ (0, false, 0) ∧
      s'.2.2.2.2.2.2.2.2.2 = s.2.2.2.2.2.2.2.2.2 + 1 ∧
      s'.2.2.2.2.2.2.2.2.1 = s.2.2.2.2.2.2.2.2.1 + 1) ∧
    (∀ {lbl s s'}, LabeledStep lbl s s' →
      s'.2.2.2.2.2.2.2.2.2 = s.2.2.2.2.2.2.2.2.2 + 1 ∧
      s'.2.2.2.2.2.2.2.2.1 = s.2.2.2.2.2.2.2.2.1 + (if 3 ≤ lbl.1 then 1 else 0)) ∧
    (∀ {s s'}, Reach Step s s' →
      s.2.2.2.2.2.2.2.2.1 ≤ s'.2.2.2.2.2.2.2.2.1 ∧
      s.2.2.2.2.2.2.2.2.2 ≤ s'.2.2.2.2.2.2.2.2.2) := by
  have hSvc : ∀ {lbl s s'}, ServiceStep lbl s s' →
      3 ≤ lbl.1 ∧ ¬(lbl.1 = 1 ∨ lbl.1 = 2) ∧ s.1 ≠ (0, false, 0) ∧
      s'.2.2.2.2.2.2.2.2.2 = s.2.2.2.2.2.2.2.2.2 + 1 ∧
      s'.2.2.2.2.2.2.2.2.1 = s.2.2.2.2.2.2.2.2.1 + 1 := by
    intro lbl s s' h
    rcases h with ⟨hp, rfl, rfl⟩ | ⟨hp, rfl, rfl⟩ |
      (⟨_, hp, _, _, rfl, rfl⟩ | ⟨hp, _, rfl, rfl⟩ | ⟨_, hp, _, _, rfl, rfl⟩) |
      (⟨_, hp, _, _, rfl, rfl⟩ | ⟨_, _, hp, _, _, rfl, rfl⟩ | ⟨_, hp, _, rfl, rfl⟩ | ⟨_, hp, _, rfl, rfl⟩) |
      (⟨hp, rfl, rfl⟩ | ⟨_, hp, rfl, rfl⟩) |
      (⟨_, hp, _, rfl, rfl⟩ | ⟨_, hp, _, rfl, rfl⟩) |
      (⟨_, hp, _, _, rfl, rfl⟩ | ⟨_, hp, _, rfl, rfl⟩ | ⟨_, hp, _, _, rfl, rfl⟩) |
      (⟨_, _, hp, _, rfl, rfl⟩ | ⟨_, hp, _, rfl, rfl⟩) |
      ⟨_, hp, rfl, rfl⟩ |
      (⟨_, _, hp, _, rfl, rfl⟩ | ⟨_, hp, _, rfl, rfl⟩) <;> simp [hp]
  have hLbl : ∀ {lbl s s'}, LabeledStep lbl s s' →
      s'.2.2.2.2.2.2.2.2.2 = s.2.2.2.2.2.2.2.2.2 + 1 ∧
      s'.2.2.2.2.2.2.2.2.1 = s.2.2.2.2.2.2.2.2.1 + (if 3 ≤ lbl.1 then 1 else 0) := by
    intro lbl s s' edge
    rcases edge with ⟨_, _, rfl, _, rfl⟩ | ⟨rfl, _, rfl⟩ | ⟨rfl, _, rfl⟩ | hs
    · simp
    · simp
    · simp
    · rcases hSvc hs with ⟨h3, _, _, he, hw⟩
      simp [h3, he, hw]
  refine ⟨hSvc, hLbl, ?_⟩
  intro s s' path
  induction path with
  | refl => exact ⟨Nat.le_refl _, Nat.le_refl _⟩
  | tail _ edge ih =>
    rcases edge with ⟨lbl, hEdge⟩
    have ⟨he, hw⟩ := hLbl hEdge
    exact ⟨by omega, by omega⟩

def ResourceShape (callers : List Nat) (s : State) : Prop :=
  let sig := baseline callers
  let out : Bool := if s.2.1 ≤ 1 then true else false
  let occCond : Prop := if s.2.1 = 0 then s.2.2.2.2.1 = false else s.2.2.2.2.1 = true
  if s.1.1 = 0 then
    s.1 = (0, false, 0) ∧ s.2.2.2.2.1 = false ∧ s.2.2.2.2.2.1 = false ∧ s.2.2.2.2.2.2.1 = sig
  else if s.1.1 = 1 then
    s.1 = (1, false, 0) ∧ s.2.2.2.2.1 = false ∧ s.2.2.2.2.2.1 = false ∧
    s.2.2.2.2.2.2.1 = sig ∧ s.2.2.2.1 = s.2.1 ∧ s.2.2.2.2.2.2.2.1 = none
  else if s.1.1 = 2 then
    s.1 = (2, false, 0) ∧ s.2.2.2.2.1 = false ∧ s.2.2.2.2.2.1 = false ∧
    s.2.2.2.2.2.2.1 = (2, 0) :: sig ∧ s.2.2.2.1 = s.2.1 ∧ s.2.2.2.2.2.2.2.1 = none
  else if s.1.1 = 3 then
    s.1 = (3, false, 0) ∧ s.2.2.2.2.2.2.1 = (3, 0) :: (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = s.2.2.2.2.1 ∧ s.2.2.2.2.2.2.2.1 = none ∧
    ((s.2.2.2.2.1 = false ∧ s.2.2.2.1 = s.2.1) ∨ (s.2.2.2.2.1 = true ∧ s.2.1 = s.2.2.2.1 + 1))
  else if s.1.1 = 4 then
    s.1 = (4, s.1.2.1, 0) ∧ s.2.2.2.2.2.2.1 = (3, 0) :: (2, 0) :: sig ∧
    s.2.2.2.2.2.2.2.1 = none ∧
    (if s.1.2.1 then
      s.2.2.2.2.2.1 = s.2.2.2.2.1 ∧ s.2.1 ≤ 1 ∧ s.2.2.2.1 = 0 ∧ occCond
    else
      s.2.2.2.2.2.1 = true ∧ s.2.2.2.2.1 = true ∧ 2 ≤ s.2.1 ∧ s.2.1 = s.2.2.2.1 + 1)
  else if s.1.1 = 5 then
    s.1 = (5, false, s.1.2.2) ∧ s.2.2.2.2.2.2.1 = (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = s.2.2.2.2.1 ∧ s.2.2.2.2.2.2.2.1 = none ∧
    s.2.1 ≤ 1 ∧ s.2.2.2.1 = 0 ∧ occCond ∧ s.1.2.2 + 1 ≤ s.2.2.1
  else if s.1.1 = 6 then
    s.1 = (6, s.1.2.1, 0) ∧ s.2.2.2.2.2.2.1 = (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = s.2.2.2.2.1 ∧ s.2.2.2.2.2.2.2.1 = none ∧
    s.1.2.1 = out ∧ occCond
  else if s.1.1 = 7 then
    s.1 = (7, s.1.2.1, 0) ∧ s.2.2.2.2.2.2.1 = (4, 0) :: (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = true ∧ s.2.2.2.2.1 = true ∧ s.2.2.2.2.2.2.2.1 = none ∧
    s.1.2.1 = out ∧ 1 ≤ s.2.1
  else if s.1.1 = 8 then
    s.1 = (8, s.1.2.1, 0) ∧ s.2.2.2.2.2.2.1 = (4, 0) :: (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = false ∧ s.2.2.2.2.1 = false ∧ s.2.2.2.2.2.2.2.1 = none ∧
    s.1.2.1 = out
  else if s.1.1 = 9 then
    s.1 = (9, s.1.2.1, 0) ∧ s.2.2.2.2.2.2.1 = (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = false ∧ s.2.2.2.2.1 = false ∧ s.2.2.2.2.2.2.2.1 = none ∧
    s.1.2.1 = out
  else
    s.1 = (10, s.1.2.1, 0) ∧ s.2.2.2.2.2.2.1 = (2, 0) :: sig ∧
    s.2.2.2.2.2.1 = false ∧ s.2.2.2.2.1 = false ∧ s.2.2.2.2.2.2.2.1 = some s.1.2.1 ∧
    s.1.2.1 = out

/-- Audit 2: Actual phase/stack/table/handle/remaining-demand invariant from initial state. -/
theorem reachable_resource_shape (callers : List Nat) :
    (∀ {lbl s s'}, ResourceShape callers s → LabeledStep lbl s s' → ResourceShape callers s') ∧
    (∀ {s}, Reach Step (initialState callers) s → ResourceShape callers s) := by
  have hStep : ∀ {lbl s s'}, ResourceShape callers s → LabeledStep lbl s s' →
      ResourceShape callers s' := by
    intro lbl s s' inv edge
    rcases edge with ⟨d, n, _, hIdle, rfl⟩ | ⟨_, _, rfl⟩ | ⟨_, _, rfl⟩ | hSvc
    · simp [ResourceShape, hIdle] at inv
      rcases inv with ⟨hOcc, hOwn, hStk⟩
      simp [ResourceShape, hOcc, hOwn, hStk]
    · exact inv
    · exact inv
    · rcases hSvc with ⟨hp, _, rfl⟩ | ⟨hp, _, rfl⟩ |
        (⟨k, hp, hRem, hFree, _, rfl⟩ | ⟨hp, hRem, _, rfl⟩ | ⟨k, hp, hRem, hOcc, _, rfl⟩) |
        (⟨rest, hp, hStack, hN, _, rfl⟩ | ⟨k, rest, hp, hStack, hN, _, rfl⟩ |
          ⟨rest, hp, hStack, _, rfl⟩ | ⟨o, hp, hWrong, _, rfl⟩) |
        (⟨hp, _, rfl⟩ | ⟨k, hp, _, rfl⟩) |
        (⟨o, hp, hOwned, _, rfl⟩ | ⟨o, hp, hOwned, _, rfl⟩) |
        (⟨o, hp, _, _, _, rfl⟩ | ⟨o, hp, hOwned, _, rfl⟩ | ⟨o, hp, _, hOcc, _, rfl⟩) |
        (⟨o, rest, hp, hStack, _, rfl⟩ | ⟨o, hp, hWrong, _, rfl⟩) |
        ⟨o, hp, _, rfl⟩ |
        (⟨o, rest, hp, hStack, _, rfl⟩ | ⟨o, hp, hWrong, _, rfl⟩)
      · simp [ResourceShape, hp] at inv ⊢; exact inv
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hOcc, hOwn, hStk, hRem, hRep⟩
        exact ⟨hStk, by rw [hOwn, hOcc], hRep, Or.inl ⟨hOcc, hRem⟩⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, _, hRep, ⟨_, hRemEq⟩ | ⟨hOccTrue, _⟩⟩
        · exact ⟨hStk, hRep, by omega⟩
        · rw [hFree] at hOccTrue; cases hOccTrue
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwn, hRep, ⟨hOcc, hRemEq⟩ | ⟨hOcc, hRemEq⟩⟩
        · have hd0 : s.2.1 = 0 := by omega
          exact ⟨hStk, hRep, hOwn, by omega, hRem, by simp [hd0, hOcc]⟩
        · have hd1 : s.2.1 = 1 := by omega
          exact ⟨hStk, hRep, hOwn, by omega, hRem, by simp [hd1, hOcc]⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwn, hRep, ⟨hOccFalse, _⟩ | ⟨_, hRemEq⟩⟩
        · rw [hOcc] at hOccFalse; cases hOccFalse
        · exact ⟨hStk, hRep, by rw [hOwn, hOcc], hOcc, by omega, hRemEq⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hRep, hOwn, hdLe, _, hOccIf⟩
        have hRest : rest = (2, 0) :: baseline callers := congrArg List.tail (hStack.symm.trans hStk)
        exact ⟨hRest, hOwn, hRep, hdLe, hOccIf⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hRep, hOwn, hdLe, hRem0, hOccIf⟩
        have hRest : rest = (2, 0) :: baseline callers := congrArg List.tail (hStack.symm.trans hStk)
        exact ⟨hRest, hOwn, hRep, hdLe, hRem0, hOccIf, by omega⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hRep, hOwn, hOcc, hdGe, _⟩
        have hRest : rest = (2, 0) :: baseline callers := congrArg List.tail (hStack.symm.trans hStk)
        have hdNotLe : ¬ s.2.1 ≤ 1 := by omega
        have hdNe0 : s.2.1 ≠ 0 := by omega
        exact ⟨hRest, by rw [hOwn, hOcc], hRep, hdNotLe, by simp [hdNe0, hOcc]⟩
      · simp [ResourceShape, hp] at inv
        rw [inv.1] at hWrong; exact False.elim (hWrong rfl)
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwn, hRep, hdLe, _, hOccIf, _⟩
        exact ⟨hStk, hOwn, hRep, hdLe, hOccIf⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwn, hRep, hdLe, hRem0, hOccIf, hkLe⟩
        exact ⟨hStk, hOwn, hRep, hdLe, hRem0, hOccIf, by omega⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwnEq, hRep, hOut, hOccIf⟩
        have hOcc : s.2.2.2.2.1 = true := by rw [← hOwnEq, hOwned]
        have hdGe1 : 1 ≤ s.2.1 := by
          by_cases hd0 : s.2.1 = 0
          · simp [hd0, hOcc] at hOccIf
          · omega
        exact ⟨hStk, hOwned, hOcc, hRep, hOut, hdGe1⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwnEq, hRep, hOut, _⟩
        exact ⟨hStk, hOwned, by rw [← hOwnEq, hOwned], hRep, hOut⟩
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, _, _, hRep, hOut, _⟩
        exact ⟨hStk, hRep, hOut⟩
      · simp [ResourceShape, hp] at inv
        rw [hOwned] at inv; cases inv.2.1
      · simp [ResourceShape, hp] at inv
        rw [hOcc] at inv; cases inv.2.2.1
      · simp [ResourceShape, hp] at inv ⊢
        rcases inv with ⟨hStk, hOwn, hOcc, hRep, hOut⟩
        exact ⟨congrArg List.tail (hStack.symm.trans hStk), hOwn, hOcc, hRep, hOut⟩
      · simp [ResourceShape, hp] at inv
        rw [inv.1] at hWrong; exact False.elim (hWrong rfl)
      · simp [ResourceShape, hp] at inv ⊢; exact inv

  intro s path

noncomputable def lastState (s : State) (steps : List ((Nat × Nat × Nat × Bool) × State)) : State :=
  steps.foldl (fun _ e => e.2) s

def ValidGlobalPath : State → List ((Nat × Nat × Nat × Bool) × State) → Prop
  | _, [] => True
  | s, (lbl, s') :: rest => LabeledStep lbl s s' ∧ ValidGlobalPath s' rest

def ValidEntryContext (ctx : List Nat × Nat × Nat × List ((Nat × Nat × Nat × Bool) × State) × State) : Prop :=
  ValidGlobalPath (initialState ctx.1) ctx.2.2.2.1 ∧
  LabeledStep (0, ctx.2.1, ctx.2.2.1, false)
    (lastState (initialState ctx.1) ctx.2.2.2.1) ctx.2.2.2.2

/-- Audit 5: Clean post-accept resources derived from global reachability and preserving acceptance. -/
theorem accepted_entries_clean
    {ctx : List Nat × Nat × Nat × List ((Nat × Nat × Nat × Bool) × State) × State}

    entry.2.2.2.2.2.2.2.2.2 = pre.2.2.2.2.2.2.2.2.2 + 1 := by
  intro pre entry sig
  have hReach : ∀ {s steps}, ValidGlobalPath s steps → Reach Step s (lastState s steps) := by
    intro s steps h
    induction steps generalizing s with
    | nil => exact Reach.refl s
    | cons hd tl ih => exact (Reach.tail (Reach.refl s) ⟨hd.1, h.1⟩).trans (ih h.2)
  have preReach := hReach hCtx.1
  have preInv := (reachable_resource_shape ctx.1).2 preReach
  rcases hCtx.2 with ⟨d', n', hEq, hIdle, hEntryEq⟩ | ⟨hEq, _⟩ | ⟨hEq, _⟩ | hSvc
  · have hdEq : ctx.2.1 = d' := congrArg (fun x => x.2.1) hEq
    have hnEq : ctx.2.2.1 = n' := congrArg (fun x => x.2.2.1) hEq

      Reach.tail preReach ⟨(0, ctx.2.1, ctx.2.2.1, false), hCtx.2⟩
    dsimp [entry, pre]
    refine ⟨entryReach, hIdle, hOcc, hOwn, hStk, ?_⟩
    rw [hEntryEq]
    exact ⟨rfl, rfl, rfl, rfl, hOcc, hOwn, hStk, rfl, rfl, rfl⟩
  · cases hEq
  · cases hEq
  · have h3 := (global_counter_accounting.1 hSvc).1; simp at h3

def ValidIntervalPrefix : State → List ((Nat × Nat × Nat × Bool) × State) → Prop
  | _, [] => True
  | s, (lbl, s') :: rest =>
      s.1 ≠ (0, false, 0) ∧ LabeledStep lbl s s' ∧ ValidIntervalPrefix s' rest

noncomputable def stateAt (s : State) : List ((Nat × Nat × Nat × Bool) × State) → Nat → State
  | [], _ => s
  | (_, _) :: _, 0 => s
  | (_, s') :: rest, i + 1 => stateAt s' rest i

noncomputable def activeWork (entry : State) (steps : List ((Nat × Nat × Nat × Bool) × State)) : Nat :=
  (lastState entry steps).2.2.2.2.2.2.2.2.1 - entry.2.2.2.2.2.2.2.2.1

noncomputable def elapsedTicks (entry : State) (steps : List ((Nat × Nat × Nat × Bool) × State)) : Nat :=
  (lastState entry steps).2.2.2.2.2.2.2.2.2 - entry.2.2.2.2.2.2.2.2.2

noncomputable def delayCount (steps : List ((Nat × Nat × Nat × Bool) × State)) : Nat :=
  steps.foldr (fun e acc => (if e.1.1 = 1 ∨ e.1.1 = 2 then 1 else 0) + acc) 0

noncomputable def peakHeap (s : State) : List ((Nat × Nat × Nat × Bool) × State) → Nat
  | [] => occupiedCount s
  | (_, s') :: rest => max (occupiedCount s) (peakHeap s' rest)

noncomputable def peakStack (s : State) : List ((Nat × Nat × Nat × Bool) × State) → Nat
  | [] => s.2.2.2.2.2.2.1.length
  | (_, s') :: rest => max s.2.2.2.2.2.2.1.length (peakStack s' rest)

/-- Audit 6: Interval work/delay/elapsed equation, including horizon 0. -/
theorem interval_tick_accounting {entry : State}
    {steps : List ((Nat × Nat × Nat × Bool) × State)}
    (h : ValidIntervalPrefix entry steps) :
    let svcCount := steps.foldr (fun e acc => (if 3 ≤ e.1.1 then 1 else 0) + acc) 0
    (lastState entry steps).2.2.2.2.2.2.2.2.1 = entry.2.2.2.2.2.2.2.2.1 + svcCount ∧
    (lastState entry steps).2.2.2.2.2.2.2.2.2 = entry.2.2.2.2.2.2.2.2.2 + steps.length ∧
    activeWork entry steps = svcCount ∧
    elapsedTicks entry steps = steps.length ∧
    elapsedTicks entry steps = activeWork entry steps + delayCount steps ∧
    activeWork entry [] = 0 ∧ elapsedTicks entry [] = 0 ∧ delayCount [] = 0 := by
  intro svcCount
  have hExact :
      (lastState entry steps).2.2.2.2.2.2.2.2.1 = entry.2.2.2.2.2.2.2.2.1 + svcCount ∧
      (lastState entry steps).2.2.2.2.2.2.2.2.2 = entry.2.2.2.2.2.2.2.2.2 + steps.length ∧
      svcCount + delayCount steps = steps.length := by
    induction steps generalizing entry with
    | nil => simp [lastState, svcCount, delayCount]
    | cons hd tl ih =>
      rcases hd with ⟨lbl, s'⟩
      rcases h with ⟨notIdle, edge, hRest⟩
      rcases ih hRest with ⟨hWk, hEl, hSum⟩
      have ⟨hStepEl, hStepWk⟩ := global_counter_accounting.2.1 edge
      have hPart : (if 3 ≤ lbl.1 then 1 else 0) + (if lbl.1 = 1 ∨ lbl.1 = 2 then 1 else 0) = 1 := by
        rcases edge with ⟨_, _, _, hIdle, _⟩ | ⟨_, hIdle, _⟩ | ⟨rfl, _⟩ | hSvc
        · exact False.elim (notIdle hIdle)
        · exact False.elim (notIdle hIdle)
        · decide
        · rcases global_counter_accounting.1 hSvc with ⟨h3, hNotDel, _⟩
          simp [h3, hNotDel]
      refine ⟨?_, ?_, ?_⟩
      · simp [lastState, svcCount] at hWk ⊢; omega
      · simp [lastState] at hEl ⊢; omega
      · simp [svcCount, delayCount] at hSum ⊢; omega
  rcases hExact with ⟨hWk, hEl, hSum⟩
  refine ⟨hWk, hEl, ?_, ?_, ?_, ?_, ?_, rfl⟩
  · dsimp [activeWork]; omega
  · dsimp [elapsedTicks]; omega
  · dsimp [activeWork, elapsedTicks]; omega
  · dsimp [activeWork, lastState]; omega
  · dsimp [elapsedTicks, lastState]; omega

noncomputable def targetL (d n : Nat) : Nat :=
  2 * min d 1 + (if d ≤ 1 then n else 0) + 8

noncomputable def remainingPotential (s : State) : Nat :=
  let a := min s.2.1 1
  let w := if s.2.1 ≤ 1 then s.2.2.1 else 0
  if s.1.1 = 0 then 0
  else if s.1.1 = 1 then 2 * a + w + 8
  else if s.1.1 = 2 then 2 * a + w + 7
  else if s.1.1 = 3 then (if s.2.2.2.2.1 then a else 2 * a) + w + 6
  else if s.1.1 = 4 then if s.1.2.1 then a + s.2.2.1 + 5 else 6
  else if s.1.1 = 5 then a + (s.1.2.2 + 1) + 4
  else if s.1.1 = 6 then (if s.2.2.2.2.2.1 then 1 else 0) + 4
  else if s.1.1 = 7 then 4
  else if s.1.1 = 8 then 3
  else if s.1.1 = 9 then 2
  else 1

theorem potential_step {callers : List Nat} {lbl : Nat × Nat × Nat × Bool} {s t : State}
    (inv : ResourceShape callers s) (notIdle : s.1 ≠ (0, false, 0))
    (edge : LabeledStep lbl s t) :
    t.2.1 = s.2.1 ∧ t.2.2.1 = s.2.2.1 ∧
    t.2.2.2.2.2.2.2.2.1 + remainingPotential t = s.2.2.2.2.2.2.2.2.1 + remainingPotential s := by
  rcases edge with ⟨_, _, _, hIdle, _⟩ | ⟨_, hIdle, _⟩ | ⟨_, _, rfl⟩ | hSvc
  · exact False.elim (notIdle hIdle)
  · exact False.elim (notIdle hIdle)
  · exact ⟨rfl, rfl, rfl⟩
  · rcases hSvc with ⟨hp, _, rfl⟩ | ⟨hp, _, rfl⟩ |
      (⟨k, hp, hRem, hFree, _, rfl⟩ | ⟨hp, hRem, _, rfl⟩ | ⟨k, hp, hRem, hOcc, _, rfl⟩) |
      (⟨rest, hp, _, hN, _, rfl⟩ | ⟨k, rest, hp, _, hN, _, rfl⟩ |
        ⟨rest, hp, _, _, rfl⟩ | ⟨o, hp, hWrong, _, rfl⟩) |
      (⟨hp, _, rfl⟩ | ⟨k, hp, _, rfl⟩) |
      (⟨o, hp, hOwned, _, rfl⟩ | ⟨o, hp, hOwned, _, rfl⟩) |
      (⟨o, hp, _, _, _, rfl⟩ | ⟨o, hp, hOwned, _, rfl⟩ | ⟨o, hp, _, hOcc, _, rfl⟩) |
      (⟨o, rest, hp, _, _, rfl⟩ | ⟨o, hp, hWrong, _, rfl⟩) |
      ⟨o, hp, _, rfl⟩ |
      (⟨o, rest, hp, _, _, rfl⟩ | ⟨o, hp, hWrong, _, rfl⟩)
    · refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp]; omega
    · simp [ResourceShape, hp] at inv
      refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, inv.1]; omega
    · simp [ResourceShape, hp] at inv
      rcases inv with ⟨_, _, _, ⟨_, hRemEq⟩ | ⟨hOccTrue, _⟩⟩
      · have ha : min s.2.1 1 = 1 := by omega
        refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hFree, ha]; omega
      · rw [hFree] at hOccTrue; cases hOccTrue
    · simp [ResourceShape, hp] at inv
      rcases inv with ⟨_, _, _, ⟨hOcc, hRemEq⟩ | ⟨hOcc, hRemEq⟩⟩
      · have hd0 : s.2.1 = 0 := by omega
        refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hOcc, hd0]; omega
      · have hd1 : s.2.1 = 1 := by omega
        refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hOcc, hd1]; omega
    · simp [ResourceShape, hp] at inv
      rcases inv with ⟨_, _, _, ⟨hOccFalse, _⟩ | ⟨_, hRemEq⟩⟩
      · rw [hOcc] at hOccFalse; cases hOccFalse
      · have hdNotLe : ¬ s.2.1 ≤ 1 := by omega
        have ha : min s.2.1 1 = 1 := by omega
        refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hOcc, hdNotLe, ha]; omega
    · simp [ResourceShape, hp] at inv
      rcases inv with ⟨_, _, hOwn, hdLe, _, hOccIf⟩
      have hOwnEq : (if s.2.2.2.2.2.1 then 1 else 0) = min s.2.1 1 := by
        rw [hOwn]
        by_cases hd0 : s.2.1 = 0
        · simp [hd0] at hOccIf; simp [hd0, hOccIf]
        · simp [hd0] at hOccIf; simp [hOccIf]; omega
      refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hN, hOwnEq]; omega
    · refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hN]; omega
    · simp [ResourceShape, hp] at inv
      rcases inv with ⟨_, _, hOwn, _⟩
      refine ⟨rfl, rfl, ?_⟩; simp [remainingPotential, hp, hOwn]; omega
    · simp [ResourceShape, hp] at inv; rw [inv.1] at hWrong; exact False.elim (hWrong rfl)

/-- Audit 9: Arbitrary mathematical context-dependent bounds transported from
`L`/`H`/`S`; completed retained occupancy zero. -/
theorem arbitrary_supplied_bound_transport
    (Bwork Bheap Bstack Bretained :
      (List Nat × Nat × Nat × List ((Nat × Nat × Nat × Bool) × State) × State) → Nat)
    {ctx : List Nat × Nat × Nat × List ((Nat × Nat × Nat × Bool) × State) × State}
    (hCtx : ValidEntryContext ctx)
    (hWorkBound : targetL ctx.2.1 ctx.2.2.1 ≤ Bwork ctx)
    (hHeapBound : min ctx.2.1 1 ≤ Bheap ctx)
    (hStackBound : (baseline ctx.1).length + 2 ≤ Bstack ctx) :
    (∀ steps, ValidIntervalPrefix ctx.2.2.2.2 steps →
      activeWork ctx.2.2.2.2 steps ≤ Bwork ctx ∧
      peakHeap ctx.2.2.2.2 steps ≤ Bheap ctx ∧
      peakStack ctx.2.2.2.2 steps ≤ Bstack ctx) ∧
    (∀ steps, ValidIntervalPrefix ctx.2.2.2.2 steps →
      (lastState ctx.2.2.2.2 steps).1 = (0, false, 0) →
      occupiedCount (lastState ctx.2.2.2.2 steps) = 0 ∧
      occupiedCount (lastState ctx.2.2.2.2 steps) ≤ Bretained ctx) := by
  constructor
  · intro steps hPref
    have hb := interval_prefix_bounds hCtx hPref
    exact ⟨Nat.le_trans hb.2.2.1 hWorkBound,
           Nat.le_trans hb.2.2.2.1 hHeapBound,
           Nat.le_trans hb.2.2.2.2.2.1 hStackBound⟩
  · intro steps hPref hIdle
    have h0 := (interval_prefix_bounds hCtx hPref).2.2.2.2.2.2 hIdle
    exact ⟨h0, by omega⟩

/-- Audit 10: A completing endpoint followed by actual arbitrary acceptance
preserves its derived clean resources. -/
theorem recovery_then_next_entry
    {ctx : List Nat × Nat × Nat × List ((Nat × Nat × Nat × Bool) × State) × State}
    (hCtx : ValidEntryContext ctx)
    {steps : List ((Nat × Nat × Nat × Bool) × State)}
    (hPrefix : ValidIntervalPrefix ctx.2.2.2.2 steps)
    (hIdle : (lastState ctx.2.2.2.2 steps).1 = (0, false, 0))
    (d' n' : Nat) :
    let completed := lastState ctx.2.2.2.2 steps
    let nextEntry : State :=
      ((1, false, 0), d', n', d', completed.2.2.2.2.1, completed.2.2.2.2.2.1,
        completed.2.2.2.2.2.2.1, none,
        completed.2.2.2.2.2.2.2.2.1, completed.2.2.2.2.2.2.2.2.2 + 1)
    let nextCtx :=
      (ctx.1, d', n',
        ctx.2.2.2.1 ++ (((0, ctx.2.1, ctx.2.2.1, false), ctx.2.2.2.2) :: steps),
        nextEntry)
    LabeledStep (0, d', n', false) completed nextEntry ∧
    ValidEntryContext nextCtx ∧
    completed.2.2.2.2.1 = false ∧
    completed.2.2.2.2.2.1 = false ∧
    completed.2.2.2.2.2.2.1 = baseline ctx.1 ∧
    nextEntry.1 = (1, false, 0) ∧
    nextEntry.2.1 = d' ∧
    nextEntry.2.2.1 = n' ∧
    nextEntry.2.2.2.1 = d' ∧
    nextEntry.2.2.2.2.1 = false ∧
    nextEntry.2.2.2.2.2.1 = false ∧
    nextEntry.2.2.2.2.2.2.1 = baseline ctx.1 ∧
    nextEntry.2.2.2.2.2.2.2.1 = none ∧
    nextEntry.2.2.2.2.2.2.2.2.1 = completed.2.2.2.2.2.2.2.2.1 ∧
    nextEntry.2.2.2.2.2.2.2.2.2 = completed.2.2.2.2.2.2.2.2.2 + 1 := by
  intro completed nextEntry nextCtx
  have hToGlobal : ∀ {s xs}, ValidIntervalPrefix s xs → ValidGlobalPath s xs := by
    intro s xs h
    induction xs generalizing s with
    | nil => exact True.intro
    | cons hd tl ih => exact ⟨h.2.1, ih h.2.2⟩
  have hAppend : ∀ {s xs ys}, ValidGlobalPath s xs → ValidGlobalPath (lastState s xs) ys →
      ValidGlobalPath s (xs ++ ys) := by
    intro s xs ys hx hy
    induction xs generalizing s with
    | nil => exact hy
    | cons hd tl ih => exact ⟨hx.1, ih hx.2 hy⟩

    Or.inl ⟨d', n', rfl, hIdle, rfl⟩
  have hPreEq : lastState (initialState ctx.1) nextCtx.2.2.2.1 = completed := by
    simp [nextCtx, lastState, List.foldl_append]
  have hNextValid : ValidEntryContext nextCtx :=
    ⟨hAppend hCtx.1 ⟨hCtx.2, hToGlobal hPrefix⟩, by rw [hPreEq]; exact hAcc⟩

  dsimp [nextCtx] at hClean

  rcases hClean with ⟨_, _, hOccPre, hOwnPre, hStkPre, hPh, hd, hn, hRem, hOcc, hOwn, hStk, hRep, hWk, hEl⟩
  exact ⟨hAcc, hNextValid, hOccPre, hOwnPre, hStkPre, hPh, hd, hn, hRem, hOcc, hOwn, hStk, hRep, hWk, hEl⟩
