/-!
Bounded certificate replay for one actual MiniRust Markdown declaration.
SourceShape is a declarative source/lexeme/group/signature relation, independent
of check. Certificate production is untrusted; reconstruction, lexical coverage,
maximality boundaries, balanced groups and unique owner/method are replayed.
The surrounding Markdown and non-body token context are fixed to the pinned
snapshot; no general Markdown/CommonMark context theorem is claimed.
This is neither a general Rust parser nor name/type/SpecR semantic resolution.
The complete source fixture below retains its upstream MIT permission notice.
-/

/-- A finite declarative universal quantifier, with a constructive decider. -/
inductive Every {A : Type} (P : A → Prop) : List A → Prop where
  | nil : Every P []
  | cons {a rest} : P a → Every P rest → Every P (a :: rest)

def everyDecision {A : Type} (P : A → Prop) (decision : ∀ a, Decidable (P a)) :
    (values : List A) → Decidable (Every P values)
  | [] => .isTrue .nil
  | a :: rest =>
      match decision a, everyDecision P decision rest with
      | .isTrue head, .isTrue tail => .isTrue (.cons head tail)
      | .isFalse denied, _ => .isFalse (fun witness => by
          cases witness with | cons head _ => exact denied head)
      | _, .isFalse denied => .isFalse (fun witness => by
          cases witness with | cons _ tail => exact denied tail)

instance {A : Type} (P : A → Prop) [DecidablePred P] (values : List A) :
    Decidable (Every P values) := everyDecision P (fun a => inferInstance) values

def Ascii (c : Char) : Prop := c.toNat < 128
def WordStart (c : Char) : Prop :=
  ('a' ≤ c ∧ c ≤ 'z') ∨ ('A' ≤ c ∧ c ≤ 'Z') ∨ c = '_'
def WordContinue (c : Char) : Prop := WordStart c ∨ ('0' ≤ c ∧ c ≤ '9')
def Digit (c : Char) : Prop := '0' ≤ c ∧ c ≤ '9'
def White (c : Char) : Prop := c = ' ' ∨ c = '\t' ∨ c = '\n' ∨ c = '\r'
def NotLineEnd (c : Char) : Prop := c ≠ '\n' ∧ c ≠ '\r'
def QuotedChar (c : Char) : Prop :=
  c ≠ '"' ∧ c ≠ '\\' ∧ c ≠ '\n' ∧ c ≠ '\r'

instance (c : Char) : Decidable (Ascii c) := by unfold Ascii; infer_instance
instance (c : Char) : Decidable (WordStart c) := by unfold WordStart; infer_instance
instance (c : Char) : Decidable (WordContinue c) := by unfold WordContinue; infer_instance
instance (c : Char) : Decidable (Digit c) := by unfold Digit; infer_instance
instance (c : Char) : Decidable (White c) := by unfold White; infer_instance
instance (c : Char) : Decidable (NotLineEnd c) := by unfold NotLineEnd; infer_instance
instance (c : Char) : Decidable (QuotedChar c) := by unfold QuotedChar; infer_instance

def Identifier (text : String) : Prop :=
  match text.toList with
  | [] => False
  | first :: rest => WordStart first ∧ Every WordContinue rest

instance (text : String) : Decidable (Identifier text) := by
  unfold Identifier
  split <;> infer_instance

inductive Mark where
  | lt | gt | colon | equal | semi | dot | comma | bang | question | bar
  | plus | minus | hash | star | amp | slash
  deriving DecidableEq, Repr

def Mark.char : Mark → Char
  | .lt => '<' | .gt => '>' | .colon => ':' | .equal => '=' | .semi => ';'
  | .dot => '.' | .comma => ',' | .bang => '!' | .question => '?' | .bar => '|'
  | .plus => '+' | .minus => '-' | .hash => '#' | .star => '*' | .amp => '&'
  | .slash => '/'

inductive Bracket where
  | brace | paren | square
  deriving DecidableEq, Repr

def Bracket.opening : Bracket → Char
  | .brace => '{' | .paren => '(' | .square => '['
def Bracket.closing : Bracket → Char
  | .brace => '}' | .paren => ')' | .square => ']'

/-- Bracket characters have no atom constructor: tree groups must account for
them. Unescaped ordinary strings and identifier lifetimes are the supported
subset; raw/byte/character strings, block comments and escapes are unsupported. -/
inductive Atom where
  | word (text : String)
  | number (text : String)
  | lifetime (text : String)
  | quoted (contents : String)
  | mark (symbol : Mark)
  deriving DecidableEq, Repr

def Atom.render : Atom → List Char
  | .word text => text.toList
  | .number text => text.toList
  | .lifetime text => '\'' :: text.toList
  | .quoted text => '"' :: (text.toList ++ ['"'])
  | .mark symbol => [symbol.char]

def Atom.Valid : Atom → Prop
  | .word text | .lifetime text => Identifier text
  | .number text => text.toList ≠ [] ∧ Every Digit text.toList
  | .quoted text => Every QuotedChar text.toList
  | .mark _ => True

instance (a : Atom) : Decidable a.Valid := by
  cases a <;> unfold Atom.Valid <;> infer_instance

inductive Token where
  | atom (value : Atom)
  | opening (bracket : Bracket)
  | closing (bracket : Bracket)
  deriving DecidableEq, Repr

def Token.render : Token → List Char
  | .atom a => a.render
  | .opening b => [b.opening]
  | .closing b => [b.closing]
def Token.Valid : Token → Prop
  | .atom a => a.Valid
  | .opening _ | .closing _ => True
instance (t : Token) : Decidable t.Valid := by
  cases t <;> unfold Token.Valid <;> infer_instance

inductive Lexeme where
  | trivia (text : String)
  | lineComment (contents : String)
  | token (value : Token)
  deriving DecidableEq, Repr

def Lexeme.render : Lexeme → List Char
  | .trivia text => text.toList
  | .lineComment text => '/' :: '/' :: text.toList
  | .token value => value.render

def Lexeme.Valid : Lexeme → Prop
  | .trivia text => text.toList ≠ [] ∧ Every White text.toList
  | .lineComment text => Every NotLineEnd text.toList
  | .token value => value.Valid

instance (l : Lexeme) : Decidable l.Valid := by
  cases l <;> unfold Lexeme.Valid <;> infer_instance

def leading : List Lexeme → Option Char
  | [] => none
  | first :: _ => first.render.head?

/-- These boundaries enforce identifier maximality, distinguish 'static from a
character literal, prevent r"..."/r#... raw forms, prevent // or /* being split
as slash punctuation, and ensure a line comment cannot consume later tokens. -/
def Lexeme.Boundary : Lexeme → Option Char → Prop
  | .token (.atom (.word _)), some c
  | .token (.atom (.number _)), some c
  | .token (.atom (.lifetime _)), some c =>
      ¬ WordContinue c ∧ c ≠ '"' ∧ c ≠ '#' ∧ c ≠ '\''
  | .token (.atom (.mark .slash)), some c => c ≠ '/' ∧ c ≠ '*'
  | .lineComment _, next => next = none ∨ next = some '\n'
  | _, _ => True

instance (l : Lexeme) (next : Option Char) : Decidable (l.Boundary next) := by
  cases l with
  | trivia text => simp only [Lexeme.Boundary]; infer_instance
  | lineComment text => simp only [Lexeme.Boundary]; infer_instance
  | token value =>
      cases value with
      | opening b => simp only [Lexeme.Boundary]; infer_instance
      | closing b => simp only [Lexeme.Boundary]; infer_instance
      | atom a =>
          cases a with
          | word text => cases next <;> simp only [Lexeme.Boundary] <;> infer_instance
          | number text => cases next <;> simp only [Lexeme.Boundary] <;> infer_instance
          | lifetime text => cases next <;> simp only [Lexeme.Boundary] <;> infer_instance
          | quoted text => simp only [Lexeme.Boundary]; infer_instance
          | mark m => cases m <;> cases next <;> simp only [Lexeme.Boundary] <;> infer_instance

inductive LexicalStream : List Lexeme → Prop where
  | nil : LexicalStream []
  | cons {first rest} : first.Valid → first.Boundary (leading rest) →
      LexicalStream rest → LexicalStream (first :: rest)

def lexicalDecision : (lexemes : List Lexeme) → Decidable (LexicalStream lexemes)
  | [] => .isTrue .nil
  | first :: rest =>
      let _ : Decidable (LexicalStream rest) := lexicalDecision rest
      decidable_of_iff
        (first.Valid ∧ first.Boundary (leading rest) ∧ LexicalStream rest)
        ⟨(fun ⟨a, b, c⟩ => .cons a b c),
          (fun witness => by cases witness with | cons a b c => exact ⟨a, b, c⟩)⟩

instance (lexemes : List Lexeme) : Decidable (LexicalStream lexemes) :=
  lexicalDecision lexemes

def renderLex (lexemes : List Lexeme) : List Char := lexemes.flatMap Lexeme.render
def tokens (lexemes : List Lexeme) : List Token :=
  lexemes.filterMap (fun l => match l with | .token t => some t | _ => none)

mutual
  inductive Tree where
    | atom (value : Atom)
    | group (kind : Bracket) (children : Forest)
  inductive Forest where
    | nil
    | cons (first : Tree) (rest : Forest)
end

mutual
  def flattenTree (tree : Tree) : List Token :=
    match tree with
    | .atom a => [.atom a]
    | .group b children => .opening b :: (flattenForest children ++ [.closing b])
  termination_by structural tree
  def flattenForest (forest : Forest) : List Token :=
    match forest with
    | .nil => []
    | .cons first rest => flattenTree first ++ flattenForest rest
  termination_by structural forest
end

def forest (trees : List Tree) : Forest :=
  trees.foldr Forest.cons .nil
def appendForest : Forest → Forest → Forest
  | .nil, right => right
  | .cons first rest, right => .cons first (appendForest rest right)
def tw (text : String) : Tree := .atom (.word text)
def tn (text : String) : Tree := .atom (.number text)
def tm (mark : Mark) : Tree := .atom (.mark mark)
def tg (kind : Bracket) (children : List Tree) : Tree := .group kind (forest children)
def lw (text : String) : Lexeme := .token (.atom (.word text))
def ln (text : String) : Lexeme := .token (.atom (.number text))
def lm (mark : Mark) : Lexeme := .token (.atom (.mark mark))
def lo (kind : Bracket) : Lexeme := .token (.opening kind)
def lc (kind : Bracket) : Lexeme := .token (.closing kind)
def lq (text : String) : Lexeme := .token (.atom (.quoted text))
def ll (text : String) : Lexeme := .token (.atom (.lifetime text))

def FencePrefix : List Char → Prop
  | '`' :: '`' :: '`' :: _ => True
  | _ => False
instance (chars : List Char) : Decidable (FencePrefix chars) := by
  unfold FencePrefix; split <;> infer_instance

/-- No prose/plain-block payload may hide another fence line. -/
inductive NoFenceLines : Bool → List Char → Prop where
  | nil (atStart) : NoFenceLines atStart []
  | cons {atStart c rest} : (atStart = true → ¬ FencePrefix (c :: rest)) →
      NoFenceLines (decide (c = '\n')) rest → NoFenceLines atStart (c :: rest)

def noFenceDecision (atStart : Bool) : (chars : List Char) →
    Decidable (NoFenceLines atStart chars)
  | [] => .isTrue (.nil atStart)
  | c :: rest =>
      let _ := noFenceDecision (decide (c = '\n')) rest
      decidable_of_iff
        ((atStart = true → ¬ FencePrefix (c :: rest)) ∧
          NoFenceLines (decide (c = '\n')) rest)
        ⟨(fun ⟨a, b⟩ => .cons a b),
          (fun witness => by cases witness with | cons a b => exact ⟨a, b⟩)⟩
instance (atStart : Bool) (chars : List Char) : Decidable (NoFenceLines atStart chars) :=
  noFenceDecision atStart chars

def lastChar : List Char → Option Char
  | [] => none
  | [c] => some c
  | _ :: rest => lastChar rest
def LineEnded (chars : List Char) : Prop := chars = [] ∨ lastChar chars = some '\n'
instance (chars : List Char) : Decidable (LineEnded chars) := by
  unfold LineEnded; infer_instance

inductive Piece where
  | prose (contents : String)
  | plainFence (contents : String)
  | rustFence (lexemes : List Lexeme) (trees : Forest)

def Piece.render : Piece → List Char
  | .prose contents => contents.toList
  | .plainFence contents => "```\n".toList ++ contents.toList ++ "```\n".toList
  | .rustFence lexemes _ => "```rust\n".toList ++ renderLex lexemes ++ "```\n".toList

def Piece.Valid : Piece → Prop
  | .prose contents | .plainFence contents =>
      NoFenceLines true contents.toList ∧ LineEnded contents.toList
  | .rustFence lexemes trees =>
      LexicalStream lexemes ∧ NoFenceLines true (renderLex lexemes) ∧
      LineEnded (renderLex lexemes) ∧ flattenForest trees = tokens lexemes
instance (piece : Piece) : Decidable piece.Valid := by
  cases piece <;> unfold Piece.Valid <;> infer_instance

def renderDocument (pieces : List Piece) : List Char := pieces.flatMap Piece.render

def ownerHeader : List Atom :=
  [.word "impl", .mark .lt, .word "T", .mark .colon, .word "Target", .mark .gt,
    .word "Memory", .word "for", .word "TreeBorrowsMemory", .mark .lt, .word "T", .mark .gt]

def stripAtoms : List Atom → Forest → Option Forest
  | [], rest => some rest
  | wanted :: suffix, .cons (.atom actual) rest =>
      if actual = wanted then stripAtoms suffix rest else none
  | _, _ => none

/-- Only top-level balanced owners in Rust fences are candidates. -/
def ownerCount : Forest → Nat
  | .nil => 0
  | current@(.cons _ rest) =>
      (match stripAtoms ownerHeader current with
        | some (.cons (.group .brace _) _) => 1
        | _ => 0) + ownerCount rest

def documentOwnerCount (pieces : List Piece) : Nat :=
  (pieces.map (fun piece => match piece with
    | .rustFence _ trees => ownerCount trees | _ => 0)).foldl (· + ·) 0

def methodCount : Forest → Nat
  | .nil => 0
  | current@(.cons _ rest) =>
      (match stripAtoms [.word "fn", .word "end_call"] current with
        | some _ => 1 | none => 0) + methodCount rest

structure BodyTerm where
  field : String
  mapper : String
  binder : String
  callee : String
  argument : String
  unitHelper : String
  propagate : Bool
  deriving DecidableEq, Repr

def BodyTerm.WellBound (body : BodyTerm) : Prop :=
  Every Identifier [body.field, body.mapper, body.binder, body.callee,
    body.argument, body.unitHelper] ∧ body.argument = body.binder
instance (body : BodyTerm) : Decidable body.WellBound := by
  unfold BodyTerm.WellBound; infer_instance

/-- The parameter and receiver names are fixed by the checked signature.
All other names are preserved. This does not resolve their semantic bindings. -/
def BodyTerm.syntax (body : BodyTerm) : Forest :=
  appendForest
    (forest [tw "extra", tm .dot, tw body.field, tm .dot, tw body.mapper,
      tg .paren [tm .bar, tw body.binder, tm .bar, tw "self", tm .dot,
        tw body.callee, tg .paren [tw body.argument]]])
    (appendForest (if body.propagate then forest [tm .question] else .nil)
      (forest [tm .semi, tw body.unitHelper, tg .paren [tg .paren []]]))

def signatureParameters : Tree :=
  tg .paren [tm .amp, tw "mut", tw "self", tm .comma, tw "extra", tm .colon,
    tw "Self", tm .colon, tm .colon, tw "FrameExtra"]

def declaration (body : BodyTerm) : Forest :=
  forest [tw "fn", tw "end_call", signatureParameters, tm .minus, tm .gt,
    tw "Result", .group .brace body.syntax]

structure Certificate where
  selectedLexemes : List Lexeme
  body : BodyTerm
  boundaries : Boundaries

def Certificate.ownerContents (cert : Certificate) : Forest :=
  appendForest pinnedBeforeMethod (appendForest (declaration cert.body) pinnedAfterMethod)
def Certificate.selectedForest (cert : Certificate) : Forest :=
  appendForest (forest (ownerHeader.map Tree.atom))
    (.cons (.group .brace cert.ownerContents) .nil)
def Certificate.document (cert : Certificate) : List Piece :=
  precedingPieces ++ [.rustFence cert.selectedLexemes cert.selectedForest] ++ followingPieces

structure Located where
  span : Span
  token : Token

def locate (cursor : Nat) : List Lexeme → List Located
  | [] => []
  | first :: rest =>
      let after := cursor + first.render.length
      match first with
      | .token token => ⟨⟨cursor, after⟩, token⟩ :: locate after rest
      | _ => locate after rest

def computedBoundaries (cert : Certificate) : Option Boundaries := do
  let fenceStart := (renderDocument precedingPieces).length
  let located := locate (fenceStart + 8) cert.selectedLexemes
  let ownerStart ← located[0]?
  let ownerLast ← located[(flattenForest cert.selectedForest).length - 1]?
  let methodIndex := ownerHeader.length + 1 + (flattenForest pinnedBeforeMethod).length
  let methodStart ← located[methodIndex]?
  let methodLast ← located[methodIndex + (flattenForest (declaration cert.body)).length - 1]?
  return
    { fence := ⟨fenceStart, fenceStart + (Piece.render (.rustFence cert.selectedLexemes cert.selectedForest)).length⟩
      owner := ⟨ownerStart.span.first, ownerLast.span.stop⟩
      method := ⟨methodStart.span.first, methodLast.span.stop⟩ }

def documentTokenCount (pieces : List Piece) : Nat :=
  (pieces.map (fun piece => match piece with
    | .rustFence lexemes _ => (tokens lexemes).length | _ => 0)).foldl (· + ·) 0

/-- Declarative, independently specified evidence. The checker is absent from
this relation. The Markdown/non-body context is closed to pinned constants.
Every source character is reconstructed, every code token is
lexically validated, bracket groups cannot be hidden in atoms, and the selected
owner/declaration/body are complete typed certificate constructors. -/
def SourceShape (source : String) (cert : Certificate) : Prop :=
  source.toList = renderDocument cert.document ∧
  Every Ascii source.toList ∧ source.toList.length ≤ 16384 ∧
  Every Piece.Valid cert.document ∧ documentTokenCount cert.document ≤ 4096 ∧
  documentOwnerCount cert.document = 1 ∧ methodCount cert.ownerContents = 1 ∧
  cert.body.WellBound ∧ computedBoundaries cert = some cert.boundaries

instance (source : String) (cert : Certificate) : Decidable (SourceShape source cert) := by
  unfold SourceShape
  infer_instance

inductive Rejection where
  | unsupportedOrInvalidCertificate
  deriving DecidableEq, Repr

def check (source : String) (cert : Certificate) : Except Rejection BodyTerm :=
  if SourceShape source cert then .ok cert.body else .error .unsupportedOrInvalidCertificate

theorem check_sound (source : String) (cert : Certificate) (body : BodyTerm)
    (checked : check source cert = .ok body) :
    SourceShape source cert ∧ body = cert.body := by
  by_cases valid : SourceShape source cert
  · have same : cert.body = body := by simpa [check, valid] using checked
    exact ⟨valid, same.symm⟩
  · simp [check, valid] at checked

theorem check_complete (source : String) (cert : Certificate)
    (shape : SourceShape source cert) : check source cert = .ok cert.body := by
  simp [check, shape]

theorem check_reconstructs (source : String) (cert : Certificate) (body : BodyTerm)
    (checked : check source cert = .ok body) :
    source.toList = renderDocument cert.document := (check_sound source cert body checked).1.1

theorem check_lexical_document (source : String) (cert : Certificate) (body : BodyTerm)
    (checked : check source cert = .ok body) : Every Piece.Valid cert.document :=
  (check_sound source cert body checked).1.2.2.2.1

theorem check_unique_owner_method (source : String) (cert : Certificate) (body : BodyTerm)
    (checked : check source cert = .ok body) :
    documentOwnerCount cert.document = 1 ∧ methodCount cert.ownerContents = 1 := by
  rcases (check_sound source cert body checked).1 with
    ⟨_, _, _, _, _, oneOwner, oneMethod, _, _⟩
  exact ⟨oneOwner, oneMethod⟩

theorem check_binder (source : String) (cert : Certificate) (body : BodyTerm)
    (checked : check source cert = .ok body) : body.argument = body.binder := by
  obtain ⟨shape, same⟩ := check_sound source cert body checked
  subst body
  rcases shape with ⟨_, _, _, _, _, _, _, bound, _⟩
  exact bound.2

theorem check_boundaries (source : String) (cert : Certificate) (body : BodyTerm)
    (checked : check source cert = .ok body) :
    computedBoundaries cert = some cert.boundaries := by
  rcases (check_sound source cert body checked).1 with
    ⟨_, _, _, _, _, _, _, _, spans⟩
  exact spans
